# Train CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement `rehearse-train`, a `chz`-based CLI that wraps `torchrun` for moshi fine-tuning.

**Architecture:** `TrainConfig` (chz) accepts a base training YAML and zero-means-no-override flags; `_merge_yaml` deep-merges overrides into the YAML and writes a temp file; `_run` shells out to `torchrun` with `PYTHONPATH` set to include `lib/moshi-finetune/` so `finetune` is importable.

**Tech Stack:** `chz`, `pyyaml`, `subprocess`, `torchrun` (PyTorch), moshi-finetune's `train.py`

import os
from pathlib import Path

import pytest
import yaml

from rehearse.train.cli import TrainConfig, _merge_yaml


@pytest.fixture
def base_yaml(tmp_path):
    cfg = {
        "run_dir": "",
        "max_steps": 100,
        "batch_size": 8,
        "duration_sec": 50.0,
        "lora": {"rank": 64, "enable": True},
    }
    p = tmp_path / "config.yaml"
    p.write_text(yaml.dump(cfg))
    return p


def test_merge_yaml_passthrough(base_yaml):
    """Base YAML values are preserved when no overrides are given."""
    tc = TrainConfig(config=base_yaml, run_dir="runs/test")
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert result["max_steps"] == 100
    assert result["batch_size"] == 8
    assert result["lora"]["rank"] == 64


def test_merge_yaml_overrides(base_yaml):
    """Non-zero override fields replace values from the base YAML."""
    tc = TrainConfig(
        config=base_yaml,
        run_dir="runs/test",
        max_steps=500,
        batch_size=4,
        lora_rank=32,
        duration_sec=25.0,
    )
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert result["max_steps"] == 500
    assert result["batch_size"] == 4
    assert result["lora"]["rank"] == 32
    assert result["duration_sec"] == 25.0


def test_merge_yaml_zero_does_not_override(base_yaml):
    """Zero values leave the base YAML untouched (zero means no override)."""
    tc = TrainConfig(config=base_yaml, run_dir="runs/test", max_steps=0, batch_size=0)
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert result["max_steps"] == 100
    assert result["batch_size"] == 8


def test_merge_yaml_run_dir_autogenerated(base_yaml):
    """run_dir is auto-generated when TrainConfig.run_dir is empty."""
    tc = TrainConfig(config=base_yaml)  # run_dir defaults to ""
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert result["run_dir"].startswith("runs/")
    assert "config" in result["run_dir"]  # contains config stem


def test_merge_yaml_run_dir_explicit(base_yaml):
    """Explicit run_dir is written into the merged YAML."""
    tc = TrainConfig(config=base_yaml, run_dir="runs/my-experiment")
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert result["run_dir"] == "runs/my-experiment"


def test_merge_yaml_wandb_key_injected(tmp_path, monkeypatch):
    """WANDB_API_KEY env var is injected into wandb.key when section exists."""
    cfg = {"run_dir": "", "wandb": {"project": "test", "key": ""}}
    p = tmp_path / "config.yaml"
    p.write_text(yaml.dump(cfg))

    monkeypatch.setenv("WANDB_API_KEY", "sk-test-123")
    tc = TrainConfig(config=p, run_dir="runs/test")
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert result["wandb"]["key"] == "sk-test-123"


def test_merge_yaml_no_wandb_key_without_section(base_yaml, monkeypatch):
    """WANDB_API_KEY is not injected when YAML has no wandb section."""
    monkeypatch.setenv("WANDB_API_KEY", "sk-test-123")
    tc = TrainConfig(config=base_yaml, run_dir="runs/test")
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert "wandb" not in result


def test_merge_yaml_output_is_valid_yaml(base_yaml):
    """Output file is valid YAML."""
    tc = TrainConfig(config=base_yaml, run_dir="runs/test")
    out = _merge_yaml(tc)
    result = yaml.safe_load(Path(out).read_text())
    assert isinstance(result, dict)

Create `rehearse/train/cli.py`:

```python
"""
Training CLI for moshi fine-tuning.

Wraps torchrun with a chz-based configuration layer. Accepts a base YAML
and override flags; merges them and delegates to lib/moshi-finetune/train.py.

── Usage ──────────────────────────────────────────────────────────────────────
    rehearse-train
    rehearse-train run_dir=runs/my-run max_steps=500 gpus=4
    rehearse-train config=rehearse/models/moshi_7B/config.yaml
    rehearse-train dry_run=true
"""
from __future__ import annotations

import os
import subprocess
import tempfile
from datetime import datetime
from pathlib import Path

import chz
import yaml

_TRAIN_PY = Path(__file__).parents[2] / "lib" / "moshi-finetune" / "train.py"

_OVERRIDES: dict[str, tuple[str, ...]] = {
    "run_dir":      ("run_dir",),
    "max_steps":    ("max_steps",),
    "batch_size":   ("batch_size",),
    "duration_sec": ("duration_sec",),
    "lora_rank":    ("lora", "rank"),
}


@chz.chz
class TrainConfig:
    config: Path = Path("rehearse/models/moshi_7B/config.yaml")
    """Base training YAML. Defines data paths, LoRA settings, optimizer params.
    Defaults to the bundled moshi-7B config."""

    run_dir: str = ""
    """Output directory for checkpoints and logs.
    Defaults to runs/<config-stem>-<YYYYMMDD-HHMMSS> if empty."""

    gpus: int = 1
    """Number of GPUs passed to torchrun --nproc-per-node.
    Use 1 for a single GPU; set to match available device count."""

    max_steps: int = 0
    """Training steps. 0 = use value from YAML (default 2000).
    Total tokens ≈ max_steps × gpus × batch_size × duration_sec × 9 × 12.5."""

    batch_size: int = 0
    """Examples per GPU per step. 0 = use value from YAML (default 16).
    Reduce if you hit OOM; also try lowering duration_sec."""

    lora_rank: int = 0
    """LoRA adapter rank. 0 = use YAML value (default 128). Keep ≤ 128."""

    duration_sec: float = 0.0
    """Max sequence length in seconds. 0 = use YAML value (default 100).
    Lowering reduces memory but may degrade quality."""

    dry_run: bool = False
    """Print the resolved torchrun command without executing."""


def _set_nested(d: dict, keys: tuple[str, ...], value: object) -> None:
    for k in keys[:-1]:
        d = d.setdefault(k, {})
    d[keys[-1]] = value


def _merge_yaml(config: TrainConfig) -> str:
    with open(config.config) as f:
        data: dict = yaml.safe_load(f) or {}

    run_dir = config.run_dir
    if not run_dir:
        stamp = datetime.now().strftime("%Y%m%d-%H%M%S")
        run_dir = f"runs/{config.config.stem}-{stamp}"

    field_values = {
        "run_dir":      run_dir,
        "max_steps":    config.max_steps,
        "batch_size":   config.batch_size,
        "duration_sec": config.duration_sec,
        "lora_rank":    config.lora_rank,
    }

    for field, keys in _OVERRIDES.items():
        value = field_values[field]
        if field == "run_dir" or value:
            _set_nested(data, keys, value)

    wandb_key = os.environ.get("WANDB_API_KEY")
    if wandb_key and "wandb" in data:
        _set_nested(data, ("wandb", "key"), wandb_key)

    with tempfile.NamedTemporaryFile(
        mode="w", suffix=".yaml", delete=False, prefix="rehearse-train-"
    ) as f:
        yaml.dump(data, f)
        return f.name


def _run(config: TrainConfig) -> None:
    temp_yaml = _merge_yaml(config)
    finetune_dir = str(_TRAIN_PY.parent)
    cmd = [
        "torchrun",
        "--nproc-per-node", str(config.gpus),
        str(_TRAIN_PY),
        temp_yaml,
    ]
    if config.dry_run:
        print(f"PYTHONPATH={finetune_dir}")
        print(" ".join(cmd))
        print(f"\nResolved config written to: {temp_yaml}")
        return
    env = {**os.environ, "PYTHONPATH": finetune_dir}
    subprocess.run(cmd, check=True, env=env)


def main() -> None:
    chz.nested_entrypoint(_run)


if __name__ == "__main__":
    main()
```
